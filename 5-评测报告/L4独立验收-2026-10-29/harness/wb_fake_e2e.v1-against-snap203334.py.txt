"""wb_fake_e2e.py — judge 独立 L4 WB runner fake 端到端 harness（阶段I）。

原则：
- 不信任 builder 自测计数；在 /tmp 快照副本上驱动**真实 run_impl / 真实共同 gate** 路径；
- 一切驱动 IO 用本 harness 的对抗性 fake（RecorderDriver/fake DB/jsonl），socket 哨兵保证零真实请求；
- fake git 仓（D/E/HEAD 三提交）喂共同 gate.verify_execution 真实 git 读取路径；
- 每个用例产 receipt JSON（含快照哈希、哨兵计数、判据），不代改 builder 文件。

用法： python3 wb_fake_e2e.py --snap /tmp/judge-l4/wb-<ts> [--out DIR]
"""

from pathlib import Path

TZ8 = dt.timezone(dt.timedelta(hours=8))


def now_iso() -> str:
    return dt.datetime.now(TZ8).isoformat(timespec="seconds")


def sha_file(p: Path) -> str:
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


def sha_bytes(b: bytes) -> str:
    return hashlib.sha256(b).hexdigest()


def jdump(p: Path, obj) -> None:
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(json.dumps(obj, ensure_ascii=False, indent=2, allow_nan=False) + "\n",
                 encoding="utf-8")


def git(*args, cwd, env_extra=None) -> subprocess.CompletedProcess:
    env = dict(os.environ, TZ="Asia/Shanghai")
    env.update(env_extra or {})
    return subprocess.run(["git", *args], cwd=str(cwd), capture_output=True, text=True, env=env)

import socket, os as _os

_orig_connect = socket.socket.connect
_orig_create = socket.create_connection
def _check(addr):
    try:
        host = addr[0]
    except Exception:
        host = str(addr)
    ok = isinstance(host, str) and (host in ("127.0.0.1", "::1", "localhost"))
    if _LOG:
        with open(_LOG, "a") as f:
            f.write("connect %s allowed=%s\n" % (host, ok))
    if not ok:
        raise AssertionError("JUDGE-SENTINEL: non-local connect refused: %r" % (addr,))
def _patched_connect(self, address):
    _check(address)
    return _orig_connect(self, address)
def _patched_create(address, *a, **k):
    _check(address)
    return _orig_create(address, *a, **k)
socket.socket.connect = _patched_connect
socket.create_connection = _patched_create
'''

class RecorderDriver:
    """独立 mock：记录全部调用；行为模式可切换（completed/active_idle/timeout_writing/
    no_trace/alias_mismatch）。零网络零进程。"""
    def __init__(self, mode="completed", fixture_dir: Path | None = None,
                 session_model="deepseek-v4.1-flash"):
        self.mode = mode
        self.calls = []
        self.fixture_dir = Path(fixture_dir) if fixture_dir else None
        self.session_model = session_model
        self.cancel_attempts = 0
        self.grow_until = None  # timeout_writing 模式下持续追加的目标文件

    def ensure_app(self):
        self.calls.append("ensure_app")
        return "mock-reused"

    def new_task(self, workspace_dir, prompt_text, model="DeepSeek V4.1-Flash",
                 title=None, port=9332, **kw):
        self.calls.append(("new_task", str(workspace_dir), model))
        return {"id": "mock-session-0001", "started_at": now_iso(),
                "model": self.session_model, "title": title or "harness"}

    def wait_completion(self, task_id, timeout_s, workspace_dir=None, **kw):
        self.calls.append(("wait_completion", task_id, timeout_s))
        if self.mode == "active_idle":
            # 模拟被 pin driver 的 active+stable 假 completed（wrapper 侧若不校验终态即中招）
            return {"status": "completed", "ended_at": now_iso(),
                    "evidence": {"db": {"id": task_id, "status": "active"},
                                 "jsonl": {"exists": True}, "stable_s": 20.1}}
        if self.mode == "timeout_writing":
            return {"status": "timeout", "ended_at": now_iso(),
                    "evidence": {"db": {"id": task_id, "status": "active"},
                                 "jsonl": {"exists": True}, "stable_s": 0.0}}
        return {"status": "completed", "ended_at": now_iso(),
                "evidence": {"db": {"id": task_id, "status": "completed",
                                    "model": self.session_model,
                                    "thought_level": "low", "mode": "craft"},
                             "jsonl": {"exists": True}, "waited_s": 0.1}}

    def collect(self, task_id, dest_dir, workspace_dir=None):
        self.calls.append(("collect", task_id, str(dest_dir)))
        dest = Path(dest_dir)
        dest.mkdir(parents=True, exist_ok=True)
        if self.mode == "no_trace":
            return {"files": []}
        # 写 fixture jsonl + session-row
        jsonl = dest / "mock-session-0001.jsonl"
        lines = []
        for o in FIXTURE_SESSION:
            o = dict(o)
            if o.get("type") == "session":
                o["model"] = self.session_model
            lines.append(json.dumps(o, ensure_ascii=False))
        if self.mode == "timeout_writing" and self.grow_until:
            # 模拟 timeout 后会话仍在写：collect 期间继续追加
            lines.append(json.dumps({"type": "late-write", "n": 1}))
        jsonl.write_text("\n".join(lines) + "\n", encoding="utf-8")
        (dest / "session-row.json").write_text(
            json.dumps({"id": task_id, "status":
                        "active" if self.mode in ("active_idle", "timeout_writing")
                        else "completed",
                        "model": self.session_model, "thought_level": "low",
                        "mode": "craft"}, ensure_ascii=False) + "\n", encoding="utf-8")
        if self.grow_until and Path(self.grow_until).exists():
            with open(self.grow_until, "a") as f:
                f.write("post-timeout write %s\n" % time.time())
        return {"files": [p.name for p in sorted(dest.iterdir())]}


# ---------------------------------------------------------------- case 执行器
def run_run_impl(fx, mode, ws_root: Path, case_root: Path, extra=None):
    """在子进程中直接调用真实 run_impl（dry=True，注入对抗 driver），返回 stdout JSON。"""
    runner = fx["WBG"] / "runner" / "l4_wb_runner.py"
    plan = json.loads((fx["L4"] / "plan.json").read_text(encoding="utf-8"))
    slot = [s for s in plan["product_slots"]
            if s["stratum"] == "wb_gateway_delivery"][0]
    harness = Path(__file__).resolve().parent
    code = f"""
import json, sys
sys.path.insert(0, {str(harness)!r})
from wb_drivers import make_driver
import importlib.util
spec = importlib.util.spec_from_file_location("l4wbr", {str(runner)!r})
m = importlib.util.module_from_spec(spec)
spec.loader.exec_module(m)
slot = json.loads({json.dumps(slot, ensure_ascii=False)!r})
drv = make_driver({mode!r})
r = m.run_impl(slot, drv, dry=True, case_root={str(case_root)!r},
               ws_root={str(ws_root)!r},
               registered_at={fx['tE']!r}, preregistration_commit={fx['E']!r},
               execution_seal_commit={fx['E']!r},
               expect_checker_sha=None)
out = {{"status": r.get("status"), "calls": [list(c) if isinstance(c, tuple) else c for c in drv.calls],
        "cancel_attempts": getattr(drv, "cancel_attempts", None),
        "meta": r.get("meta"), "accounting": r.get("accounting")}}
print("@@RESULT@@" + json.dumps(out, ensure_ascii=False))
"""
    lib = fx["REPO"].parent / "sentinel-lib"
    write_sentinel(lib)
    log = fx["REPO"].parent / "sentinel.log"
    env = dict(os.environ, HOME=str(fx["home"]), PYTHONPATH=str(lib),
               JUDGE_SENTINEL_LOG=str(log), PYTHONDONTWRITEBYTECODE="1")
    p = subprocess.run([sys.executable, "-c", code], capture_output=True, text=True,
                       env=env, timeout=600)
    marker = "@@RESULT@@"
    raw = None
    for line in (p.stdout or "").splitlines():
        if line.startswith(marker):
            raw = json.loads(line[len(marker):])
    return {"rc": p.returncode, "stdout_tail": (p.stdout or "")[-1500:],
            "stderr_tail": (p.stderr or "")[-2500:], "result": raw}


def read_result(r):
    return r.get("result") or {}


def main() -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("--snap", required=True)
    ap.add_argument("--out", default=None)
    ap.add_argument("--only", default=None, help="逗号分隔用例过滤")
    a = ap.parse_args()
    snap = Path(a.snap).resolve()
    out_dir = Path(a.out) if a.out else (
        Path(__file__).resolve().parent / "results" / snap.name)
    work_root = Path("/tmp/judge-l4/fixture-" + snap.name.split("-", 1)[-1])
    if work_root.exists():
        shutil.rmtree(work_root)
    work_root.mkdir(parents=True)
    harness = Path(__file__).resolve().parent
    shutil.copy2(harness / "wb_drivers.py", work_root / "wb_drivers.py")
    fx = build_fixture(snap, work_root)
    cases = {}
    only = set(a.only.split(",")) if a.only else None

    def rec(name, verdict, detail):
        cases[name] = {"verdict": verdict, "detail": detail}

    def run_case(name):
        return (only is None) or (name in only)

    case_root = fx["CASE"] / "bird-ca-schools-15"

    def clean_runs():
        shutil.rmtree(fx["WBG"] / "runs", ignore_errors=True)

    # 哨兵日志预置（显式零尝试）
    (work_root / "sentinel.log").touch()
    (work_root / "sentinel-w1.log").touch()
    (work_root / "sentinel-w5.log").touch()

    # ---- W1 E 未放行：cmd_run_slot 真实子进程必须拒绝且零驱动 ----
    if run_case("W1_e_seal_block"):
        lib = work_root / "sentinel-lib"
        write_sentinel(lib)
        log = work_root / "sentinel-w1.log"
        env = dict(os.environ, HOME=str(fx["home"]), PYTHONPATH=str(lib),
                   JUDGE_SENTINEL_LOG=str(log), PYTHONDONTWRITEBYTECODE="1")
        fx["binding"](False)
        runner = fx["WBG"] / "runner" / "l4_wb_runner.py"
        p = subprocess.run([sys.executable, str(runner), "run-slot",
                            "l4-20261029-wb-gateway-bird-ca-schools-15-r1"],
                           capture_output=True, text=True, env=env, timeout=300,
                           cwd=str(fx["REPO"]))
        refused = (fx["WBG"] / "receipts" / "release-refused.json")
        rec("W1_e_seal_block", "pass" if (p.returncode == 2 and refused.exists()) else "fail",
            {"rc": p.returncode, "stdout_tail": p.stdout[-400:], "stderr_tail": p.stderr[-400:],
             "refused_receipt": refused.exists(),
             "gate_reasons": (json.loads(refused.read_text()).get("gate", {})
                              .get("reasons", []) if refused.exists() else None)})

    # ---- W2 干跑基线（对抗 driver=completed 正常路径）----
    if run_case("W2_dry_baseline"):
        clean_runs()
        fx["binding"](True)
        ws = work_root / "ws-w2"
        r = run_run_impl(fx, "completed", ws, case_root)
        res = read_result(r)
        meta = (res or {}).get("meta") or {}
        acc = (res or {}).get("accounting") or {}
        ok = (res and res.get("status") == "ok" and acc.get("execution_state") == "executed"
              and meta.get("registered_at") == fx["tE"]
              and meta.get("usage", {}).get("per_call")
              and len(meta["usage"]["per_call"]) == 2)
        rec("W2_dry_baseline", "pass" if ok else "fail",
            {"rc": r["rc"], "status": (res or {}).get("status"),
             "accounting_state": acc.get("execution_state"),
             "registered_at": meta.get("registered_at"),
             "usage_per_call_n": len(meta.get("usage", {}).get("per_call", [])),
             "stderr_tail": r["stderr_tail"][-500:]})

    # ---- W3 alias 不匹配 → run_impl 后真实 gate --admission 必须实拒 ----
    if run_case("W3_alias_gate"):
        clean_runs()
        ws = work_root / "ws-w3"
        r = run_run_impl(fx, "alias_mismatch", ws, case_root)
        res = read_result(r)
        meta = (res or {}).get("meta") or {}
        verdict_alias = ((meta.get("alias_evidence") or {}).get("verdict"))
        # 真实共同 gate admission（组件级独立复算）
        gspec = importlib.util.spec_from_file_location(
            "fxgate", fx["REPO"] / "1-评测方案" / "工具链" / "l4-execution" / "gate.py")
        gmod = importlib.util.module_from_spec(gspec)
        gspec.loader.exec_module(gmod)
        bp = fx["binding"](True)
        try:
            adm = gmod.gate_admission(str(bp), "workbuddy", repo=fx["REPO"],
                                      plan_dir=fx["L4"])
            adm_out = json.dumps(adm, ensure_ascii=False, default=str)[:4000]
        except Exception as e:
            adm_out = {"error": repr(e)[:800]}
        rec("W3_alias_gate", "evidence",
            {"runner_verdict_string": verdict_alias,
             "runner_accounting_state": ((res or {}).get("accounting") or {}).get("execution_state"),
             "runner_score_recorded": ((res or {}).get("accounting") or {}).get("score"),
             "gate_admission": adm_out,
             "note": "判据：alias 不匹配须令 formal_admission=false 且原因可见；"
                     "若 gate 通过或未提 alias → gate 侧缺陷"})

    # ---- W4 missing trace 不得 SHA(empty)+clean 冒真证据 ----
    if run_case("W4_no_trace"):
        clean_runs()
        ws = work_root / "ws-w4"
        r = run_run_impl(fx, "no_trace", ws, case_root)
        res = read_result(r)
        meta = (res or {}).get("meta") or {}
        empty_sha = hashlib.sha256(b"").hexdigest()
